cmake_minimum_required(VERSION 3.21)
project(HEG_VST_Bridge VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This prototype targets Windows; cross-compilation via MinGW-w64 is supported experimentally.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    add_executable(HEGWorker32 worker32/main.cpp)
    target_compile_definitions(HEGWorker32 PRIVATE UNICODE _UNICODE)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_executable(HEGController64 controller64/main.cpp)
    target_compile_definitions(HEGController64 PRIVATE UNICODE _UNICODE)
else()
    message(FATAL_ERROR "Unsupported architecture")
endif()

if(MINGW)
    add_link_options(-static-libgcc -static-libstdc++)
    if(TARGET HEGWorker32)
        target_link_options(HEGWorker32 PRIVATE -municode)
    endif()
    if(TARGET HEGController64)
        target_link_options(HEGController64 PRIVATE -municode)
    endif()
endif()
